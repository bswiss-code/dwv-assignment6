compelvars <- c("year", "helpfulv", "trustv")

#Account for NA answers
compelfiltereddata <- compelfiltereddata |>
  mutate(
    helpfulv_f = as_factor(helpfulv),                            
    helpfulv_f = fct_na_value_to_level(helpfulv_f, level = "NA")
  )
#Recode and create a line graph showing the proportion of people who identify as helpful or trustworthy over the years
compel <- compelfiltereddata |>
  mutate(
    helpful_r = case_when(helpfulv == 1 ~ 1,
                          helpfulv %in% 2:3 ~ 0),     
    trust_r   = case_when(trustv == 1 ~ 1,            
                          trustv %in% 2:3 ~ 0)    
  )

yearly_compel <- compel |>
  group_by(year) |>
  summarise(
    Helpful = mean(helpful_r, na.rm = TRUE),
    Trust   = mean(trust_r,   na.rm = TRUE)
  ) |>
  pivot_longer(-year, names_to = "measure", values_to = "proportion") |>
  filter(!is.nan(proportion))     

ggplot(yearly_compel, aes(x = year, y = proportion, linetype = measure)) +
  geom_line() +
  scale_y_continuous(limits = c(0, .5)) +
  labs(x = NULL, y = "Proportion", linetype = NULL) +
  theme_classic()
